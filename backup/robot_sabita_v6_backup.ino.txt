#define N_NODE 6
#define NO_EDGE 999.0

const char NODE_NAMES[N_NODE] = {'A', 'B', 'C', 'D', 'E', 'F'};

float distMatrix[N_NODE][N_NODE] = {
  { 0.00,  1.07,  NO_EDGE, NO_EDGE, 1.51,  1.21 },
  { 1.07,  0.00,  1.80,  NO_EDGE, NO_EDGE, 1.16 },
  { NO_EDGE, 1.80,  0.00,  1.68,  NO_EDGE, 1.41 },
  { NO_EDGE, NO_EDGE, 1.68,  0.00,  1.49,  1.07 },
  { 1.51,  NO_EDGE, NO_EDGE, 1.49,  0.00,  1.54 },
  { 1.21,  1.16,  1.41,  1.07,  1.54,  0.00 }
};

const float ALPHA = 1.0;
const float BETA  = 2.0;
const float RHO   = 0.3;
const int   N_ANTS = 6;
const int   N_ITER = 100;
const float TAU0  = 1.0;

float pheromone[N_NODE][N_NODE];
int   bestRoute[N_NODE + 1];
float bestLength = 999999.0;

int nodeIndexFromChar(char c) {
  c = toupper(c);
  for (int i = 0; i < N_NODE; i++) {
    if (NODE_NAMES[i] == c) return i;
  }
  return -1;
}

char nodeCharFromIndex(int idx) {
  if (idx < 0 || idx >= N_NODE) return '?';
  return NODE_NAMES[idx];
}

void initPheromone() {
  for (int i = 0; i < N_NODE; i++)
    for (int j = 0; j < N_NODE; j++)
      pheromone[i][j] = TAU0;
}

bool constructRoute(int startNode, int route[]) {
  bool visited[N_NODE];
  for (int i = 0; i < N_NODE; i++) visited[i] = false;

  route[0] = startNode;
  visited[startNode] = true;
  int current = startNode;

  int step = 1;
  int maxTries = 100;
  int tries = 0;

  while (step < N_NODE && tries < maxTries) {
    tries++;
    float scores[N_NODE];
    float totalScore = 0.0;
    bool hasCandidate = false;

    for (int j = 0; j < N_NODE; j++) {
      if (!visited[j] && distMatrix[current][j] < NO_EDGE) {
        float tau = pow(pheromone[current][j], ALPHA);
        float eta = pow(1.0 / distMatrix[current][j], BETA);
        scores[j] = tau * eta;
        totalScore += scores[j];
        hasCandidate = true;
      } else {
        scores[j] = 0.0;
      }
    }

    if (!hasCandidate) {
      float bestFallbackDist = NO_EDGE;
      int bestFallback = -1;
      for (int j = 0; j < N_NODE; j++) {
        if (distMatrix[current][j] < NO_EDGE && j != current) {
          for (int k = 0; k < N_NODE; k++) {
            if (!visited[k] && distMatrix[j][k] < NO_EDGE) {
              if (distMatrix[current][j] < bestFallbackDist) {
                bestFallbackDist = distMatrix[current][j];
                bestFallback = j;
              }
              break;
            }
          }
        }
      }
      if (bestFallback == -1) return false;
      current = bestFallback;
      continue;
    }

    float r = ((float)random(0, 10000) / 10000.0) * totalScore;
    float cumulative = 0.0;
    int chosen = -1;
    for (int j = 0; j < N_NODE; j++) {
      if (scores[j] > 0) {
        cumulative += scores[j];
        if (cumulative >= r) { chosen = j; break; }
      }
    }
    if (chosen == -1) {
      for (int j = 0; j < N_NODE; j++) {
        if (scores[j] > 0) { chosen = j; break; }
      }
    }

    route[step] = chosen;
    visited[chosen] = true;
    current = chosen;
    step++;
  }

  return (step >= N_NODE);
}

float routeLength(int route[]) {
  float total = 0.0;
  for (int i = 0; i < N_NODE - 1; i++) {
    total += distMatrix[route[i]][route[i + 1]];
  }
  total += distMatrix[route[N_NODE - 1]][route[0]];
  return total;
}

void updatePheromoneACO(int allRoutes[][N_NODE], float allLengths[], int nRoutes) {
  for (int i = 0; i < N_NODE; i++) {
    for (int j = 0; j < N_NODE; j++) {
      pheromone[i][j] *= (1.0 - RHO);
      if (pheromone[i][j] < 0.0001) pheromone[i][j] = 0.0001;
    }
  }
  for (int k = 0; k < nRoutes; k++) {
    float deposit = 1.0 / allLengths[k];
    for (int i = 0; i < N_NODE - 1; i++) {
      int a = allRoutes[k][i];
      int b = allRoutes[k][i + 1];
      pheromone[a][b] += deposit;
      pheromone[b][a] += deposit;
    }
    int a = allRoutes[k][N_NODE - 1];
    int b = allRoutes[k][0];
    pheromone[a][b] += deposit;
    pheromone[b][a] += deposit;
  }
}

bool runACO(int startNode) {
  initPheromone();
  bestLength = 999999.0;
  bool foundAny = false;

  int allRoutes[N_ANTS][N_NODE];
  float allLengths[N_ANTS];

  for (int iter = 0; iter < N_ITER; iter++) {
    int validCount = 0;
    for (int ant = 0; ant < N_ANTS; ant++) {
      int route[N_NODE];
      bool ok = constructRoute(startNode, route);
      if (ok) {
        float len = routeLength(route);
        allLengths[validCount] = len;
        for (int i = 0; i < N_NODE; i++) allRoutes[validCount][i] = route[i];
        validCount++;
        foundAny = true;
        if (len < bestLength) {
          bestLength = len;
          for (int i = 0; i < N_NODE; i++) bestRoute[i] = route[i];
          bestRoute[N_NODE] = startNode;
        }
      }
    }
    if (validCount > 0) updatePheromoneACO(allRoutes, allLengths, validCount);
    yield();
  }
  return foundAny;
}

// ===================== MOTOR =====================
// Konvensi BTS7960 (hasil kalibrasi lapangan): RPWM=0 & LPWM=speed -> roda MAJU
//                                               RPWM=speed & LPWM=0 -> roda MUNDUR
void wheelR(bool forward, int speed) {
  if (forward) { ledcWrite(CH_R_RPWM, 0);     ledcWrite(CH_R_LPWM, speed); }
  else         { ledcWrite(CH_R_RPWM, speed); ledcWrite(CH_R_LPWM, 0);     }
}
void wheelL(bool forward, int speed) {
  if (forward) { ledcWrite(CH_L_RPWM, 0);     ledcWrite(CH_L_LPWM, speed); }
  else         { ledcWrite(CH_L_RPWM, speed); ledcWrite(CH_L_LPWM, 0);     }
}
void motorStop() {
  ledcWrite(CH_R_RPWM, 0); ledcWrite(CH_R_LPWM, 0);
  ledcWrite(CH_L_RPWM, 0); ledcWrite(CH_L_LPWM, 0);
}
void motorMaju() {
  wheelR(true, MOTOR_SPEED);
  wheelL(true, MOTOR_SPEED);
}
void motorMundur() {
  wheelR(false, MOTOR_SPEED);
  wheelL(false, MOTOR_SPEED);
}
void motorKanan() {
  wheelR(false, MOTOR_SPEED); // roda kanan mundur
  wheelL(true, MOTOR_SPEED);  // roda kiri maju -> pivot ke kanan
}
void motorKiri() {
  wheelR(true, MOTOR_SPEED);  // roda kanan maju
  wheelL(false, MOTOR_SPEED); // roda kiri mundur -> pivot ke kiri
}

// ===================== BACA ADC DENGAN OVERSAMPLING =====================
int readADCAvg(int pin) {
  long sum = 0;
  for (int i = 0; i < ADC_OVERSAMPLE; i++) {
    sum += analogRead(pin);
    delayMicroseconds(100);
  }
  return (int)(sum / ADC_OVERSAMPLE);
}

// ===================== BACA SEMUA SENSOR & KONVERSI KE DIGITAL =====================
// Hitam (garis) = ADC rendah = 1 (HIGH), Putih (background) = ADC tinggi = 0 (LOW)
void readSensors() {
  for (int i = 0; i < 5; i++) {
    int val = readADCAvg(sensorPin[i]);
    sDigital[i] = (val < threshold[i]) ? 1 : 0;
  }
}

// ===================== KALIBRASI OTOMATIS SAAT BOOT =====================
// Fase 1 (2.5 detik pertama): taruh robot di atas PUTIH
// Fase 2 (2.5 detik berikutnya): taruh robot di atas HITAM
void runCalibration() {
  Serial.println("\n=== KALIBRASI SENSOR ===");
  Serial.println("Fase 1: Taruh robot di atas GARIS PUTIH (2.5 detik)...");
  String calibWhiteMsg = "{\"type\":\"state\",\"state\":\"CALIBRATING_WHITE\"}";
  webSocket.broadcastTXT(calibWhiteMsg);

  // Fase 1: kalibrasi putih
  int samplesW[5] = {0, 0, 0, 0, 0};
  int countW = 0;
  unsigned long t0 = millis();
  while (millis() - t0 < CALIB_DURATION_MS / 2) {
    for (int i = 0; i < 5; i++) samplesW[i] += readADCAvg(sensorPin[i]);
    countW++;
    delay(50);
    yield();
  }
  for (int i = 0; i < 5; i++) whiteVal[i] = samplesW[i] / countW;
  Serial.print("Putih: ");
  for (int i = 0; i < 5; i++) { Serial.print(whiteVal[i]); Serial.print(" "); }
  Serial.println();

  // Fase 2: kalibrasi hitam
  Serial.println("Fase 2: Taruh robot di atas GARIS HITAM (2.5 detik)...");
  String calibBlackMsg = "{\"type\":\"state\",\"state\":\"CALIBRATING_BLACK\"}";
  webSocket.broadcastTXT(calibBlackMsg);

  int samplesB[5] = {0, 0, 0, 0, 0};
  int countB = 0;
  unsigned long t1 = millis();
  while (millis() - t1 < CALIB_DURATION_MS / 2) {
    for (int i = 0; i < 5; i++) samplesB[i] += readADCAvg(sensorPin[i]);
    countB++;
    delay(50);
    yield();
  }
  for (int i = 0; i < 5; i++) blackVal[i] = samplesB[i] / countB;
  Serial.print("Hitam: ");
  for (int i = 0; i < 5; i++) { Serial.print(blackVal[i]); Serial.print(" "); }
  Serial.println();

  // Hitung threshold = midpoint putih dan hitam
  Serial.print("Threshold: ");
  for (int i = 0; i < 5; i++) {
    threshold[i] = (whiteVal[i] + blackVal[i]) / 2;
    Serial.print(threshold[i]); Serial.print(" ");
  }
  Serial.println();

  calibDone = true;
  Serial.println("=== KALIBRASI SELESAI ===");
  String calibDoneMsg = "{\"type\":\"state\",\"state\":\"WAIT_START\"}";
  webSocket.broadcastTXT(calibDoneMsg);
}

/*
 * STRATEGI LINE-FOLLOW (relatif badan robot, TANPA orientasi global):
 *
 * Mode NORMAL (default, dipakai saat belum 5 node luar selesai):
 *   - Prioritas LURUS dulu jika S3 aktif & tidak ada percabangan besar
 *   - Koreksi ringan kiri/kanan untuk tetap di atas garis
 *   - Jika ada PERSIMPANGAN (banyak sensor aktif sekaligus, indikasi cabang),
 *     robot tetap pilih jalur yang paling LURUS/MELENGKUNG (mengikuti
 *     kontur luar pentagon), BUKAN cabang tajam ke dalam (menuju F).
 *     Asumsi: cabang ke F secara fisik lebih tajam/berbeda sudut
 *     dibanding kelengkungan pentagon luar.
 *
 * Mode SEEK_CENTER (aktif setelah 5 node luar selesai dikunjungi):
 *   - Robot PRIORITASKAN BELOK (ambil cabang manapun yang tersedia,
 *     bukan lurus) untuk mencari jalur masuk ke F.
 *   - Strategi: begitu ada indikasi persimpangan, ambil belok
 *     (default ke kanan, bisa ditukar ke kiri jika hasil test lapangan
 *     menunjukkan F lebih sering di sisi kiri robot).
 */

bool seekCenterMode = false; // diaktifkan dari FSM setelah 5 node luar selesai
bool forceFirstTurnRight = false; // aktif sekali saat start dari F, untuk ambil 1 dari 5 cabang

void lineFollowStep() {
  // S1=kiri jauh, S2=kiri tengah, S3=tengah, S4=kanan tengah, S6=kanan jauh
  int s1 = sDigital[0]; // GPIO33
  int s2 = sDigital[1]; // GPIO32
  int s3 = sDigital[2]; // GPIO35
  int s4 = sDigital[3]; // GPIO34
  int s6 = sDigital[4]; // GPIO39 (S6, bukan S5)

  bool isJunction = (s1 + s2 + s3 + s4 + s6) >= 3;

  if (forceFirstTurnRight) {
    if (isJunction) {
      motorKanan();
      return;
    } else {
      forceFirstTurnRight = false;
    }
  }

  if (seekCenterMode && isJunction) {
    motorKanan();
    return;
  }

  // Mode normal: ikuti garis, prioritas LURUS
  if (s3 == 1 && s2 == 0 && s4 == 0) {
    // Garis tepat tengah -> lurus
    motorMaju();
  } else if (s2 == 1 && s3 == 1 && s4 == 0) {
    // Garis sedikit kiri -> koreksi kanan ringan
    wheelR(true, MOTOR_SPEED / 2);
    wheelL(true, MOTOR_SPEED);
  } else if (s3 == 1 && s4 == 1 && s2 == 0) {
    // Garis sedikit kanan -> koreksi kiri ringan
    wheelR(true, MOTOR_SPEED);
    wheelL(true, MOTOR_SPEED / 2);
  } else if (s2 == 1 && s1 == 0 && s3 == 0) {
    // Garis agak kiri
    wheelR(true, MOTOR_SPEED / 3);
    wheelL(true, MOTOR_SPEED);
  } else if (s4 == 1 && s6 == 0 && s3 == 0) {
    // Garis agak kanan
    wheelR(true, MOTOR_SPEED);
    wheelL(true, MOTOR_SPEED / 3);
  } else if (s1 == 1) {
    // Garis jauh kiri -> belok kiri tajam
    motorKiri();
  } else if (s6 == 1) {
    // Garis jauh kanan -> belok kanan tajam
    motorKanan();
  } else if (s3 == 1 && s2 == 1 && s4 == 1) {
    // Persimpangan lebar -> tetap lurus
    motorMaju();
  } else {
    // Tidak ada garis -> tetap maju pelan
    motorMaju();
  }
}

// ===================== DFPLAYER =====================
void playAudioForNode(int nodeIdx) {
  if (!dfPlayerReady) return;
  if (nodeIdx < 0 || nodeIdx >= N_NODE) return;
  int fileNum = nodeIdx + 1; // A=1, B=2, ... F=6
  dfPlayer.play(fileNum);
  Serial.println("Play audio: " + String(namaKarya[nodeIdx]) + " (Node " + String(NODE_NAMES[nodeIdx]) + ")");

  String json = "{\"type\":\"dfp\",\"status\":\"playing\",\"file\":" + String(fileNum) +
                ",\"name\":\"" + String(namaKarya[nodeIdx]) +
                "\",\"node\":\"Node " + String(NODE_NAMES[nodeIdx]) +
                "\",\"desc\":\"" + String(deskKarya[nodeIdx]) + "\"}";
  webSocket.broadcastTXT(json);
}

void stopAudio() {
  if (dfPlayerReady) {
    dfPlayer.stop();
    String stopMsg = "{\"type\":\"dfp\",\"status\":\"stopped\"}";
    webSocket.broadcastTXT(stopMsg);
  }
}

// ===================== QR PARSING =====================
// Extract huruf node dari "Node X, Art NAMA, Audio XXX"
int getNodeFromQR(String qrData) {
  qrData.trim();
  String lower = qrData;
  lower.toLowerCase();
  int idx = lower.indexOf("node ");
  if (idx >= 0) {
    int charPos = idx + 5;
    if (charPos < (int)qrData.length()) {
      char c = toupper(qrData.charAt(charPos));
      return nodeIndexFromChar(c);
    }
  }
  return -1;
}

}

// ===================== WEBSOCKET EVENT =====================
void webSocketEvent(uint8_t num, WStype_t type, uint8_t* payload, size_t length) {
  if (type == WStype_DISCONNECTED) {
    // Jangan auto-stop motor di sini supaya misi otomatis tidak terganggu disconnect WS
  }
  else if (type == WStype_CONNECTED) {
    String json = "{\"type\":\"dfp\",\"status\":\"" + String(dfPlayerReady ? "ready" : "error") +
                  "\",\"volume\":" + String(currentVolume) + "}";
    webSocket.sendTXT(num, json);
  }
  else if (type == WStype_TEXT) {
    String msg = String((char*)payload);

    // Manual override (mode testing manual, hanya aktif kalau robot WAIT_START)
    if (robotState == STATE_WAIT_START) {
      if      (msg == "MAJU")   motorMaju();
      else if (msg == "MUNDUR") motorMundur();
      else if (msg == "KANAN")  motorKanan();
      else if (msg == "KIRI")   motorKiri();
      else if (msg == "STOP")   motorStop();
    }

    if (msg.startsWith("SPEED:")) {
      MOTOR_SPEED = constrain(msg.substring(6).toInt(), 0, 255);
    }
    else if (msg.startsWith("PLAY:")) {
      int nodeIdx = msg.substring(5).toInt() - 1;
      playAudioForNode(nodeIdx);
    }
    else if (msg == "AUDIO_STOP") {
      stopAudio();
    }
    else if (msg.startsWith("VOL:")) {
      currentVolume = constrain(msg.substring(4).toInt(), 0, VOLUME_MAX);
      if (dfPlayerReady) dfPlayer.volume(currentVolume);
    }
    else if (msg == "RESET_MISSION") {
      resetMission();
    }
    else if (msg == "START_MISSION") {
      // Tombol MULAI dari web - hanya aktif saat STATE_READY
      if (robotState == STATE_READY) {
        Serial.println("Tombol MULAI ditekan. Memulai ACO...");
        robotState = STATE_COMPUTING_ACO;
        broadcastState("COMPUTING");
      } else {
        Serial.println("START_MISSION diabaikan, state bukan READY");
      }
    }
    else if (msg == "RECALIB") {
      // Kalibrasi ulang sensor - hanya bisa saat WAIT_START atau READY
      if (robotState == STATE_WAIT_START || robotState == STATE_READY) {
        Serial.println("Memulai kalibrasi ulang dari web...");
        runCalibration();
      } else {
        Serial.println("Kalibrasi hanya bisa dilakukan saat WAIT_START");
        String notReadyMsg = "{\"type\":\"state\",\"state\":\"WAIT_START\"}";
        webSocket.broadcastTXT(notReadyMsg);
      }
    }
  }
}

// ===================== SETUP =====================
void setup() {
  WRITE_PERI_REG(RTC_CNTL_BROWN_OUT_REG, 0);
  esp_task_wdt_init(30, false);

  Serial.begin(115200);
  Serial.println("\n=== SABITA Robot v6 - FULL INTEGRATION ===");

  randomSeed(analogRead(0)); // seed random untuk ACO

  // --- Motor PWM ---
  ledcSetup(CH_R_RPWM, PWM_FREQ, PWM_RES);
  ledcSetup(CH_R_LPWM, PWM_FREQ, PWM_RES);
  ledcSetup(CH_L_RPWM, PWM_FREQ, PWM_RES);
  ledcSetup(CH_L_LPWM, PWM_FREQ, PWM_RES);
  ledcAttachPin(MOT_R_RPWM, CH_R_RPWM);
  ledcAttachPin(MOT_R_LPWM, CH_R_LPWM);
  ledcAttachPin(MOT_L_RPWM, CH_L_RPWM);
  ledcAttachPin(MOT_L_LPWM, CH_L_LPWM);
  motorStop();
  Serial.println("Motor OK");

  // --- Line Follower ADC ---
  adc_power_acquire();          // fix glitch GPIO36 & GPIO39 saat WiFi aktif (+1mA)
  analogReadResolution(12);       // 12-bit ADC (0-4095)
  analogSetAttenuation(ADC_11db); // range 0-3.3V
  pinMode(PIN_CLP, INPUT);
  Serial.println("Line Follower ADC OK (12-bit, 0-3.3V)");

  // --- Kalibrasi sensor (blocking, sebelum WiFi supaya tidak ganggu ADC) ---
  Serial.println("Memulai kalibrasi sensor...");
  // Kalibrasi sementara pakai nilai default dulu (WiFi belum up, webSocket belum jalan)
  // Kalibrasi ulang bisa dipanggil dari web setelah WiFi up via RESET_MISSION
  // Untuk saat ini, baca nilai awal sebagai estimasi putih
  for (int i = 0; i < 5; i++) {
    long sum = 0;
    for (int j = 0; j < 20; j++) { sum += analogRead(sensorPin[i]); delay(10); }
    whiteVal[i] = sum / 20;
    blackVal[i] = whiteVal[i] / 4; // estimasi hitam = 25% dari putih
    threshold[i] = (whiteVal[i] + blackVal[i]) / 2;
  }
  Serial.print("Threshold awal: ");
  for (int i = 0; i < 5; i++) { Serial.print(threshold[i]); Serial.print(" "); }
  Serial.println();
  Serial.println("(Kalibrasi ulang: taruh di putih lalu hitam setelah terhubung ke web)");


  // --- GM67 UART2 ---
  GM67Serial.begin(9600, SERIAL_8N1, GM67_TX, GM67_RX);
  Serial.println("GM67 OK");

  // --- DFPlayer UART1 ---
  DFPSerial.begin(9600, SERIAL_8N1, DFP_RX, DFP_TX);
  Serial.println("DFPlayer init...");
  delay(2000);
  if (dfPlayer.begin(DFPSerial, false, true)) {
    dfPlayerReady = true;
    dfPlayer.volume(currentVolume);
    Serial.println("DFPlayer OK");
  } else {
    dfPlayerReady = false;
    Serial.println("DFPlayer GAGAL");
  }

  // --- WiFi AP ---
  WiFi.softAP(AP_SSID, AP_PASS);
  Serial.println("IP: " + WiFi.softAPIP().toString());

  // --- Web Server ---
  server.on("/", []() { server.send_P(200, "text/html", HTML_PAGE); });
  server.begin();

  // --- WebSocket ---
  webSocket.begin();
  webSocket.onEvent(webSocketEvent);

  // --- Reset state awal ---
  for (int i = 0; i < N_NODE; i++) visitedNode[i] = false;

  Serial.println("\n=== READY - Menunggu scan QR start ===");
  Serial.println("http://" + WiFi.softAPIP().toString());
}

// ===================== LOOP =====================
void loop() {
  server.handleClient();
  webSocket.loop();

  // --- Baca sensor ADC + konversi ke digital ---
  readSensors(); // isi sDigital[0..4]: S1,S2,S3,S4,S6
  int s1 = sDigital[0]; // GPIO33
  int s2 = sDigital[1]; // GPIO32
  int s3 = sDigital[2]; // GPIO35
  int s4 = sDigital[3]; // GPIO34
  int s6 = sDigital[4]; // GPIO39 (S6)
  // Baca nilai ADC raw untuk monitoring di web
  int rawVal[5];
  for (int i = 0; i < 5; i++) rawVal[i] = readADCAvg(sensorPin[i]);

  if (millis() - lastSensorUpdate >= SENSOR_INTERVAL) {
    lastSensorUpdate = millis();

    String arah = "TIDAK ADA GARIS";
    if (s1==1&&s2==1&&s3==1&&s4==1&&s6==1)      arah = "PERSIMPANGAN";
    else if (s3==1&&s2==0&&s4==0)                arah = "LURUS";
    else if (s2==1&&s3==1&&s4==0)                arah = "BELOK KIRI";
    else if (s3==1&&s4==1&&s2==0)                arah = "BELOK KANAN";
    else if (s1==1)                              arah = "KIRI TAJAM";
    else if (s6==1)                              arah = "KANAN TAJAM";

    String json = "{\"type\":\"sensor\","
      "\"s1\":" + String(s1) + ",\"s2\":" + String(s2) + ",\"s3\":" + String(s3) +
      ",\"s4\":" + String(s4) + ",\"s6\":" + String(s6) +
      ",\"arah\":\"" + arah +
      "\",\"r1\":" + String(rawVal[0]) +
      ",\"r2\":" + String(rawVal[1]) +
      ",\"r3\":" + String(rawVal[2]) +
      ",\"r4\":" + String(rawVal[3]) +
      ",\"r5\":" + String(rawVal[4]) +
      ",\"thr\":" + String(threshold[2]) + "}";
    webSocket.broadcastTXT(json);
  }

  // --- Baca QR (selalu jalan, perilaku tergantung state) ---
  String qrData = "";
  if (GM67Serial.available()) {
    unsigned long t = millis();
    while (millis() - t < 200) {
      if (GM67Serial.available()) {
        char c = GM67Serial.read();
        if (c == '\r' || c == '\n') {
          if (qrData.length() > 0) break;
        } else {
          qrData += c;
        }
        t = millis();
      }
    }
    if (qrData.length() > 0) {
      Serial.println("QR: " + qrData);
      String qrJson = "{\"type\":\"qr\",\"data\":\"" + qrData + "\"}";
      webSocket.broadcastTXT(qrJson);
    }
  }

  // ============================================================
  //  FSM ROBOT
  // ============================================================
  switch (robotState) {

    case STATE_WAIT_START: {
      motorStop();
      if (qrData.length() > 0) {
        int nodeIdx = getNodeFromQR(qrData);
        if (nodeIdx >= 0) {
          startNodeIdx = nodeIdx;
          Serial.println("Start node terdeteksi: Node " + String(NODE_NAMES[nodeIdx]));
          Serial.println("Menunggu tombol MULAI dari web...");
          robotState = STATE_READY;
          broadcastState("READY");
          // Kirim info node ke web
          String info = "{\"type\":\"startnode\",\"node\":\"" + String(NODE_NAMES[nodeIdx]) + "\"}";
          webSocket.broadcastTXT(info);
        }
      }
      break;
    }

    case STATE_READY: {
      motorStop(); // diam, tunggu tombol MULAI
      // Tetap baca QR kalau ada update node
      if (qrData.length() > 0) {
        int nodeIdx = getNodeFromQR(qrData);
        if (nodeIdx >= 0 && nodeIdx != startNodeIdx) {
          // Scan QR node berbeda = update start node
          startNodeIdx = nodeIdx;
          Serial.println("Start node diupdate: Node " + String(NODE_NAMES[nodeIdx]));
          String info = "{\"type\":\"startnode\",\"node\":\"" + String(NODE_NAMES[nodeIdx]) + "\"}";
          webSocket.broadcastTXT(info);
        }
      }
      break;
    }

    case STATE_COMPUTING_ACO: {
      Serial.println("Menghitung ACO dari start=" + String(NODE_NAMES[startNodeIdx]) + "...");
      unsigned long t0 = millis();
      bool ok = runACO(startNodeIdx);
      unsigned long t1 = millis();
      Serial.println("ACO selesai dalam " + String(t1 - t0) + " ms");

      if (ok) {
        Serial.print("Rute: ");
        for (int i = 0; i <= N_NODE; i++) {
          Serial.print(NODE_NAMES[bestRoute[i]]);
          if (i < N_NODE) Serial.print("-");
        }
        Serial.println(" | " + String(bestLength, 2) + " m");

        for (int i = 0; i < N_NODE; i++) visitedNode[i] = false;
        visitedNode[startNodeIdx] = true;
        nodesVisitedCount = 1;
        routeProgress = 1;

        // --- Tentukan mode awal line-follow berdasar titik start ---
        bool startIsCenter = (startNodeIdx == 5); // index 5 = 'F'
        if (startIsCenter) {
          forceFirstTurnRight = true;
          seekCenterMode = false;
          Serial.println("Start dari F (tengah): paksa belok kanan di persimpangan pertama");
        } else {
          forceFirstTurnRight = false;
          seekCenterMode = false;
          Serial.println("Start dari node luar: mode NORMAL (ikuti pentagon, cari F setelah 5 node)");
        }

        broadcastRouteStatus();
        robotState = STATE_MOVING;
        broadcastState("MOVING");
      } else {
        Serial.println("ACO GAGAL menemukan rute! Cek topologi graf.");
        robotState = STATE_WAIT_START;
        broadcastState("WAIT_START");
      }
      break;
    }

    case STATE_MOVING: {
      lineFollowStep(); // sDigital[] sudah diisi readSensors() di atas

      if (qrData.length() > 0) {
        int nodeIdx = getNodeFromQR(qrData);
        if (nodeIdx >= 0 && !visitedNode[nodeIdx]) {
          motorStop();
          Serial.println("Sampai di Node " + String(NODE_NAMES[nodeIdx]));
          visitedNode[nodeIdx] = true;
          nodesVisitedCount++;
          playAudioForNode(nodeIdx);
          arrivedTimestamp = millis();
          broadcastRouteStatus();
          robotState = STATE_ARRIVED;
          broadcastState("ARRIVED");
        }
      }
      break;
    }

    case STATE_ARRIVED: {
      motorStop();
      if (millis() - arrivedTimestamp >= AUDIO_PLAY_DURATION) {

        // Hitung berapa node LUAR (bukan F) yang sudah dikunjungi
        int outerVisitedCount = 0;
        for (int i = 0; i < N_NODE; i++) {
          if (i != 5 && visitedNode[i]) outerVisitedCount++; // index 5 = F
        }

        if (nodesVisitedCount >= N_NODE) {
          // semua 6 node selesai, kembali ke start
          Serial.println("Semua node selesai. Kembali ke Node " + String(NODE_NAMES[startNodeIdx]));
          routeProgress = N_NODE;
          seekCenterMode = false; // perjalanan pulang pakai mode normal lagi
          robotState = STATE_MOVING;
          broadcastState("MOVING");
        } else if (outerVisitedCount >= 5 && !visitedNode[5]) {
          // 5 node luar sudah selesai, F belum dikunjungi -> aktifkan mode cari pusat
          Serial.println("5 node luar selesai. Mode SEEK_CENTER diaktifkan untuk mencari F.");
          seekCenterMode = true;
          routeProgress++;
          robotState = STATE_MOVING;
          broadcastState("MOVING");
        } else {
          routeProgress++;
          robotState = STATE_MOVING;
          broadcastState("MOVING");
        }
      }
      break;
    }

    case STATE_FINISHED: {
      motorStop();
      // diam di sini sampai di-reset manual lewat web
      break;
    }
  }

  // Deteksi selesai total: robot balik ke start node setelah semua node dikunjungi
  if (robotState == STATE_MOVING && nodesVisitedCount >= N_NODE && routeProgress >= N_NODE) {
    if (qrData.length() > 0) {
      int nodeIdx = getNodeFromQR(qrData);
      if (nodeIdx == startNodeIdx) {
        motorStop();
        Serial.println("Robot kembali ke start. MISI SELESAI.");
        robotState = STATE_FINISHED;
        broadcastState("FINISHED");
      }
    }
  }
}
